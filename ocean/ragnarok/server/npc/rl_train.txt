//===== Training worlds only (local script, not part of rAthena) =====
// @rlreset <task> turns the caller into a fresh character for one episode of that task: alive with
// full HP and SP, no status effects, the task's job, levels, stats, skills and gear, then a random
// cell of the task's map. train/world.sh loads this only into throwaway training worlds, where every
// bot may use it. Tasks (the env's RG_TASK_* in PufferLib/ocean/ragnarok/ragnarok.h):
//   0  field leveling: Novice, levels 1/1, Main Gauche, Cotton Shirt; prt_fild08
//   1  Swordman skills: Swordman, levels 25/25, STR 50 AGI 30 VIT 30 DEX 20, Bash 10, Magnum
//      Break 5, Sword Mastery 10, Increase HP Recovery 5, Blade, Padded Armor; pay_fild01 (Spores)
//   2  survival: a weaker Swordman (levels 18/18, STR 35 AGI 20 VIT 15 DEX 15, the same skills and
//      gear, 10 Red Potions) on pay_fild02 among Wolves (level 25), which help each other
//   3  Mage skills: Mage, levels 25/25, INT 50 DEX 35 VIT 25 AGI 20, Fire, Cold and Lightning Bolt 5,
//      Soul Strike 3, Napalm Beat 3, Fire Ball 1, Frost Diver 1, Increase SP Recovery 3, Rod, Silk
//      Robe; mjolnir_01 (mostly earth monsters, weak to fire: Coco, Caramel, Horn; some wind)
//   4  Archer skills: Archer, levels 25/25, DEX 50 AGI 35 VIT 25 STR 20, Double Strafe 10, Arrow
//      Shower 5, Owl's Eye 5, Vulture's Eye 5, Bow and 2,000 Arrows, Cotton Shirt; prt_fild03 (Yoyo,
//      Smokie)
//   5  Acolyte skills: Acolyte, levels 25/25, STR 35 VIT 35 INT 40 DEX 20, Heal 5, Blessing 5,
//      Increase AGI 5, Divine Protection 5, Demon Bane 5, Club, Silk Robe; pay_fild01
//   6  Thief skills: Thief, levels 25/25, STR 40 AGI 50 DEX 30 VIT 10, Double Attack 10, Improve
//      Dodge 5, Envenom 5, Steal 3, Hiding 2, Main Gauche, Padded Armor; prt_fild03
//   7  Merchant skills: Merchant, levels 25/25, STR 50 VIT 35 DEX 30 AGI 15, Mammonite 10 and
//      200,000 zeny to pay for it, Enlarge Weight Limit 5, Discount 3, Overcharge 3, Axe, Padded
//      Armor; mjolnir_01
// Each gets 30 Red Potions (survival 10). $@rl_maps$ lists every task's map for the other training
// scripts (rl_spread.txt, rl_census.txt); RG_TASK_MAPS in slurm/world.sh is the same list.
-	script	rl_train	-1,{
OnInit:
	setarray $@rl_maps$[0], "prt_fild08", "pay_fild01", "pay_fild02", "mjolnir_01", "prt_fild03";
	bindatcmd "rlreset", strnpcinfo(3) + "::OnReset", 0, 99;
	end;

OnReset:
	.@task = atoi(.@atcmd_parameters$[0]);
	recovery 0;
	sc_end SC_ALL;
	clearitem;
	// Per task: job, base and job level, stats (STR AGI VIT INT DEX LUK; none for the Novice, who
	// spends the points a reset gives by the env's fixed rule), skills (id, level), weapon and armor.
	.@level = 25;
	.@potions = 30;
	switch (.@task) {
	case 1:
		.@job = Job_Swordman;
		setarray .@stat[0], 50, 30, 30, 1, 20, 1;
		setarray .@skill[0], 5, 10, 7, 5, 2, 10, 4, 5;
		setarray .@gear[0], 1109, 2312;
		.@map$ = "pay_fild01";
		break;
	case 2:
		// Weaker than the wolves, so pulling a pack, or fighting on at low HP, gets it killed.
		.@job = Job_Swordman;
		.@level = 18;
		.@potions = 10;
		setarray .@stat[0], 35, 20, 15, 1, 15, 1;
		setarray .@skill[0], 5, 10, 7, 5, 2, 10, 4, 5;
		setarray .@gear[0], 1109, 2312;
		.@map$ = "pay_fild02";
		break;
	case 3:
		.@job = Job_Mage;
		setarray .@stat[0], 1, 20, 25, 50, 35, 1;
		setarray .@skill[0], 19, 5, 14, 5, 20, 5, 13, 3, 11, 3, 17, 1, 15, 1, 9, 3;
		setarray .@gear[0], 1601, 2321;
		.@map$ = "mjolnir_01";
		break;
	case 4:
		.@job = Job_Archer;
		setarray .@stat[0], 20, 35, 25, 1, 50, 1;
		setarray .@skill[0], 46, 10, 47, 5, 43, 5, 44, 5;
		setarray .@gear[0], 1701, 2301;
		.@map$ = "prt_fild03";
		break;
	case 5:
		.@job = Job_Acolyte;
		setarray .@stat[0], 35, 1, 35, 40, 20, 1;
		setarray .@skill[0], 28, 5, 34, 5, 29, 5, 22, 5, 23, 5;
		setarray .@gear[0], 1501, 2321;
		.@map$ = "pay_fild01";
		break;
	case 6:
		.@job = Job_Thief;
		setarray .@stat[0], 40, 50, 10, 1, 30, 1;
		setarray .@skill[0], 48, 10, 49, 5, 52, 5, 50, 3, 51, 2;
		setarray .@gear[0], 1207, 2312;
		.@map$ = "prt_fild03";
		break;
	case 7:
		.@job = Job_Merchant;
		setarray .@stat[0], 50, 15, 35, 1, 30, 1;
		setarray .@skill[0], 42, 10, 36, 5, 37, 3, 38, 3;
		setarray .@gear[0], 1301, 2312;
		.@map$ = "mjolnir_01";
		break;
	default:
		.@job = Job_Novice;
		.@level = 1;
		setarray .@gear[0], 1207, 2301;
		.@map$ = "prt_fild08";
		break;
	}
	if (Class != .@job)
		jobchange .@job;
	resetlvl 1;
	resetskill;
	BaseLevel = .@level;
	JobLevel = .@level;
	if (getarraysize(.@stat)) {
		statusup2 bStr, .@stat[0] - 1;
		statusup2 bAgi, .@stat[1] - 1;
		statusup2 bVit, .@stat[2] - 1;
		statusup2 bInt, .@stat[3] - 1;
		statusup2 bDex, .@stat[4] - 1;
		statusup2 bLuk, .@stat[5] - 1;
		StatusPoint = 0;
	}
	SkillPoint = 0;
	for (.@i = 0; .@i < getarraysize(.@skill); .@i += 2)
		skill .@skill[.@i], .@skill[.@i + 1], 0;
	for (.@i = 0; .@i < getarraysize(.@gear); .@i++) {
		getitem .@gear[.@i], 1;
		equip .@gear[.@i];
	}
	if (.@job == Job_Archer) {
		getitem 1750, 2000;
		equip 1750;
	}
	if (.@job == Job_Merchant)
		Zeny = 200000;
	getitem 501, .@potions;
	percentheal 100, 100;
	warp .@map$, 0, 0;
	end;
}
