//===== Training worlds only (local script, not part of rAthena) =====
// Every 10 minutes of game time, write to the map server's log each task map's monster count and its
// monsters in a 4x4 grid of 100x100-cell squares (does the map stay evenly stocked?), and the Jellopy
// lying on the training field. The maps are $@rl_maps$ from rl_train.txt. Under dynamic_mobs a map
// nobody is on counts 0.
-	script	rl_census	-1,{
OnInit:
	initnpctimer;
	end;
OnTimer600000:
	.@line$ = "rl_census";
	for (.@m = 0; .@m < getarraysize($@rl_maps$); .@m++) {
		.@map$ = $@rl_maps$[.@m];
		.@line$ += " " + .@map$ + " " + mobcount(.@map$, "all") + " grid";
		for (.@gy = 0; .@gy < 4; .@gy++)
			for (.@gx = 0; .@gx < 4; .@gx++)
				.@line$ += " " + getareaunits(BL_MOB, .@map$, .@gx * 100, .@gy * 100, .@gx * 100 + 99, .@gy * 100 + 99);
	}
	debugmes .@line$ + " jellopy_on_prt_fild08 " + getareadropitem("prt_fild08", 0, 0, 399, 399, 909);
	initnpctimer;
	end;
}
